# StarHermit engine script game: one package, played natively by the
# StarHermit client and in browsers by the engine's web runtime. Browsers open
# launch=index.html, the loader `npm run package` writes (it pins the engine
# on wiki.starhermit.com by hash; never edit it). Re-run `npm run package`
# after every change and commit index.html + starhermit-files.json.
launch=index.html
engine.api=1
engine.entry=native/main.ts
engine.version=1.19.1
engine.minor=20
engine.capabilities=platform,storage,clipboard

# Keyboard controls (players rebind keys and pads in the runtime's Options).
# The crate is also played with the pointer or touch: press on the stall to
# aim, drag to move the aim, release to drop; the aim slider and every
# action's button do the same.
engine.group.aimleft=Aim and drop
control.aimleft=ArrowLeft+KeyA | Aim left
control.aimright=ArrowRight+KeyD | Aim right
control.drop=Space+Enter+NumpadEnter | Drop fruit
engine.group.undo=Run
control.undo=KeyU | Undo (practice)
control.hint=KeyH | Hint
control.restart=KeyR | Restart run
control.pause=KeyP+Escape | Pause / resume

# Gamepad defaults (the browser build had none): the D-pad or left stick
# aims, South drops, North undoes, West hints, Select restarts, Start or East
# pauses.
engine.gamepad.aimleft=DpadLeft+LeftStickLeft
engine.gamepad.aimright=DpadRight+LeftStickRight
engine.gamepad.drop=South
engine.gamepad.undo=North
engine.gamepad.hint=West
engine.gamepad.restart=Select
engine.gamepad.pause=Start+East
